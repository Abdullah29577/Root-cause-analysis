import pandas as pd
from scipy.stats import chi2_contingency

# ↓↓↓ CHANGE THIS to the folder where your Olist CSVs live ↓↓↓
DATA_DIR = r"C:\Users\basith\Downloads\archive"

# 1. Load the three files we need
orders  = pd.read_csv(f"{DATA_DIR}/olist_orders_dataset.csv")
items   = pd.read_csv(f"{DATA_DIR}/olist_order_items_dataset.csv")
reviews = pd.read_csv(f"{DATA_DIR}/olist_order_reviews_dataset.csv")

# 2. Count how many DISTINCT sellers each order had
sellers_per_order = items.groupby("order_id")["seller_id"].nunique().reset_index()
sellers_per_order.columns = ["order_id", "seller_count"]

# 3. Take one review score per order (some orders have duplicates)
reviews_one = reviews.groupby("order_id")["review_score"].max().reset_index()

# 4. Join it all together
df = sellers_per_order.merge(reviews_one, on="order_id")

# 5. Make the two categories
df["seller_group"] = df["seller_count"].apply(lambda x: "multi" if x > 1 else "single")
df["review_group"] = df["review_score"].apply(lambda x: "bad" if x <= 2 else "good")

# 6. Build the 2x2 table
table = pd.crosstab(df["seller_group"], df["review_group"])
print("Counts:")
print(table)
print()

# 7. Show the bad-review RATE for each group (the effect size that matters)
rates = df.groupby("seller_group")["review_group"].apply(lambda s: (s == "bad").mean())
print("Bad-review rate:")
print((rates * 100).round(1).astype(str) + "%")
print()

# 8. Run the chi-square test
chi2, p, dof, expected = chi2_contingency(table)
print(f"chi-square = {chi2:.1f}")
print(f"p-value    = {p:.2e}")
print(f"dof        = {dof}")
